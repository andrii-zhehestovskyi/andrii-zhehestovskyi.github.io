Code sample: metamorphic test for an LLM

Illustrative snippet. A paraphrased question must not change the factual answer — a core idea in Project P2.

import pytest

PAIRS = [
    ("What year did the UK GDPR take effect?",
     "In which year did UK GDPR come into force?"),
]

@pytest.mark.parametrize("q1,q2", PAIRS)
def test_paraphrase_invariance(model, q1, q2):
    a1, a2 = model.ask(q1), model.ask(q2)
    # same extracted fact, not identical wording
    assert extract_year(a1) == extract_year(a2)
